# M3 — GI 基类 + IBL 实现计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 引入 `IGITechnique` 抽象基类，并以 IBL（split-sum + BRDF LUT）作为第一个具体实现，把当前 forward 中的常量 `ambient` 替换为基于环境贴图的间接漫反射 + 环境镜面反射。

**Architecture:**
- 启动期一次性烘焙：HDR equirectangular → cubemap → diffuse irradiance cubemap + 多 mip GGX prefiltered cubemap + 2D BRDF LUT，全部 compute pipeline 完成。
- IBL 资源以 Set=1 描述符提供给 forward 着色器（Set=0 仍是场景/材质）。
- `IGITechnique` 持有该 Set 的 layout、Set 自身、以及对应着色器变体名；ForwardPass 依据当前技术构建对应 graphics pipeline。
- M3 只接入 IBL；ImGui 下拉框只有 "IBL" 一项，但接入位置已经是未来切换的标准位置。

**Tech Stack:** Vulkan 1.3 (compute pipelines, dynamic rendering), slang, stb_image (HDR), glm.

**验证标准（来自设计文档 §9）:** env map IBL 正确 — 在 Sponza 中关闭直接光后仍能看到环境亮度差与法线方向反射；金属高粗糙球（人为放一个测试球或修改材质 metallic=1）能看到模糊环境反射；漫反射环境色随 N 方向变化。

```cpp
#include "env_loader.h"
#include <stb_image.h>  // STB_IMAGE_IMPLEMENTATION 已在 gltf_loader.cpp 定义
#include <cmath>

namespace somegi {

bool loadHdrEquirect(const std::filesystem::path& path, EnvCpu& env, std::string& errOut) {
    int w, h, c;
    float* pix = stbi_loadf(path.string().c_str(), &w, &h, &c, 3);
    if (!pix) {
        errOut = std::string("stbi_loadf failed: ") + path.string();
        return false;
    }
    env.width = w; env.height = h;
    env.rgbaF32.resize(size_t(w) * h * 4);
    for (int i = 0; i < w * h; ++i) {
        env.rgbaF32[i * 4 + 0] = pix[i * 3 + 0];
        env.rgbaF32[i * 4 + 1] = pix[i * 3 + 1];
        env.rgbaF32[i * 4 + 2] = pix[i * 3 + 2];
        env.rgbaF32[i * 4 + 3] = 1.0f;
    }
    stbi_image_free(pix);
    errOut.clear();
    return true;
}

void makeFallbackSky(EnvCpu& env) {
    env.width = 512; env.height = 256;
    env.rgbaF32.resize(size_t(env.width) * env.height * 4);
    for (int y = 0; y < env.height; ++y) {
        float t = float(y) / float(env.height - 1);  // 0=top, 1=bottom
        // 顶部偏蓝、地平线偏白、地面暗
        float r, g, b;
        if (t < 0.5f) {
            float k = t * 2.0f;
            r = 0.4f + 0.5f * k;
            g = 0.6f + 0.3f * k;
            b = 1.0f;
        } else {
            float k = (t - 0.5f) * 2.0f;
            r = 0.9f - 0.7f * k;
            g = 0.9f - 0.7f * k;
            b = 1.0f - 0.7f * k;
        }
        for (int x = 0; x < env.width; ++x) {
            int i = (y * env.width + x) * 4;
            env.rgbaF32[i + 0] = r;
            env.rgbaF32[i + 1] = g;
            env.rgbaF32[i + 2] = b;
            env.rgbaF32[i + 3] = 1.0f;
        }
    }
}

}
```

```hlsl
// shaders/gi/ibl/equi_to_cube.slang
[[vk::binding(0, 0)]] Texture2D<float4>  gEquirect;
[[vk::binding(1, 0)]] SamplerState        gLinear;
[[vk::binding(2, 0)]] RWTexture2DArray<float4> gCubeFaces;  // viewType=2DArray, layerCount=6

[[vk::push_constant]] struct {
    uint cubeSize;   // 边长（一面）
    uint _p0, _p1, _p2;
} gPC;

static const float PI = 3.14159265359;

float3 dirForFaceUV(uint face, float2 uv) {
    // uv ∈ [0,1]，s,t 转为 [-1,1]
    float2 st = uv * 2.0 - 1.0;
    float3 d;
    switch (face) {
        case 0: d = float3( 1.0, -st.y, -st.x); break;  // +X
        case 1: d = float3(-1.0, -st.y,  st.x); break;  // -X
        case 2: d = float3( st.x,  1.0,  st.y); break;  // +Y
        case 3: d = float3( st.x, -1.0, -st.y); break;  // -Y
        case 4: d = float3( st.x, -st.y,  1.0); break;  // +Z
        default:d = float3(-st.x, -st.y, -1.0); break;  // -Z
    }
    return normalize(d);
}

float2 dirToEquirect(float3 d) {
    float phi = atan2(d.z, d.x);          // [-PI, PI]
    float theta = asin(clamp(d.y, -1.0, 1.0));  // [-PI/2, PI/2]
    float u = phi / (2.0 * PI) + 0.5;
    float v = 0.5 - theta / PI;
    return float2(u, v);
}

[shader("compute")]
[numthreads(8, 8, 1)]
void cs_main(uint3 dt : SV_DispatchThreadID) {
    if (dt.x >= gPC.cubeSize || dt.y >= gPC.cubeSize || dt.z >= 6) return;
    float2 uv = (float2(dt.xy) + 0.5) / float(gPC.cubeSize);
    float3 dir = dirForFaceUV(dt.z, uv);
    float2 eq = dirToEquirect(dir);
    float3 c = gEquirect.SampleLevel(gLinear, eq, 0).rgb;
    gCubeFaces[uint3(dt.xy, dt.z)] = float4(c, 1.0);
}
```

```cpp
Image uploadEquirect(Device& d, VkCommandPool pool, const EnvCpu& env) {
    ImageDesc id{};
    id.format = VK_FORMAT_R32G32B32A32_SFLOAT;
    id.extent = {(uint32_t)env.width, (uint32_t)env.height, 1};
    id.usage = VK_IMAGE_USAGE_TRANSFER_DST_BIT | VK_IMAGE_USAGE_SAMPLED_BIT;
    Image img(d, id);

    Buffer staging(d, env.rgbaF32.size() * sizeof(float),
                   VK_BUFFER_USAGE_TRANSFER_SRC_BIT,
                   VK_MEMORY_PROPERTY_HOST_VISIBLE_BIT | VK_MEMORY_PROPERTY_HOST_COHERENT_BIT);
    std::memcpy(staging.mapped(), env.rgbaF32.data(), env.rgbaF32.size() * sizeof(float));

    oneShotSubmit(d, pool, [&](VkCommandBuffer cmd) {
        std::vector<VkImageMemoryBarrier2> bs;
        bs.push_back(imgBarrier(img.image(),
            VK_IMAGE_LAYOUT_UNDEFINED, VK_IMAGE_LAYOUT_TRANSFER_DST_OPTIMAL,
            VK_PIPELINE_STAGE_2_TOP_OF_PIPE_BIT, 0,
            VK_PIPELINE_STAGE_2_COPY_BIT, VK_ACCESS_2_TRANSFER_WRITE_BIT,
            0, 1, 0, 1));
        pipelineBarrier(cmd, bs);

        VkBufferImageCopy c{};
        c.imageSubresource = {VK_IMAGE_ASPECT_COLOR_BIT, 0, 0, 1};
        c.imageExtent = id.extent;
        vkCmdCopyBufferToImage(cmd, staging.handle(), img.image(),
```

```cpp
struct ComputeKernel {
    Device* d = nullptr;
    VkDescriptorSetLayout dsl = VK_NULL_HANDLE;
    VkPipelineLayout pl = VK_NULL_HANDLE;
    VkPipeline pipe = VK_NULL_HANDLE;

    void destroy() {
        if (!d) return;
        if (pipe) vkDestroyPipeline(d->device(), pipe, nullptr);
        if (pl)   vkDestroyPipelineLayout(d->device(), pl, nullptr);
        if (dsl)  vkDestroyDescriptorSetLayout(d->device(), dsl, nullptr);
        pipe = VK_NULL_HANDLE; pl = VK_NULL_HANDLE; dsl = VK_NULL_HANDLE;
    }
};

ComputeKernel makeKernel(Device& d, const std::filesystem::path& spvPath,
                         const std::vector<VkDescriptorSetLayoutBinding>& bindings,
                         uint32_t pushConstantBytes) {
    ComputeKernel k; k.d = &d;

    VkDescriptorSetLayoutCreateInfo li{VK_STRUCTURE_TYPE_DESCRIPTOR_SET_LAYOUT_CREATE_INFO};
    li.bindingCount = (uint32_t)bindings.size();
    li.pBindings    = bindings.data();
    VK_CHECK(vkCreateDescriptorSetLayout(d.device(), &li, nullptr, &k.dsl));

    VkPushConstantRange pc{};
    pc.stageFlags = VK_SHADER_STAGE_COMPUTE_BIT;
    pc.size = pushConstantBytes;

    VkPipelineLayoutCreateInfo plci{VK_STRUCTURE_TYPE_PIPELINE_LAYOUT_CREATE_INFO};
    plci.setLayoutCount = 1; plci.pSetLayouts = &k.dsl;
    if (pushConstantBytes > 0) { plci.pushConstantRangeCount = 1; plci.pPushConstantRanges = &pc; }
    VK_CHECK(vkCreatePipelineLayout(d.device(), &plci, nullptr, &k.pl));

    ShaderModule cs(d, spvPath);
    VkPipelineShaderStageCreateInfo ss{VK_STRUCTURE_TYPE_PIPELINE_SHADER_STAGE_CREATE_INFO};
    ss.stage = VK_SHADER_STAGE_COMPUTE_BIT;
    ss.module = cs.handle();
    ss.pName = "cs_main";

    VkComputePipelineCreateInfo cpi{VK_STRUCTURE_TYPE_COMPUTE_PIPELINE_CREATE_INFO};
    cpi.stage = ss; cpi.layout = k.pl;
    VK_CHECK(vkCreateComputePipelines(d.device(), VK_NULL_HANDLE, 1, &cpi, nullptr, &k.pipe));
    return k;
}

VkDescriptorPool makeOneShotDescPool(Device& d, uint32_t maxSets,
```

## Task 7: prefilter_diffuse.slang + 调度

Lambertian 漫反射卷积：对法线 N 半球积分 cos-weighted env，简单 Riemann 采样足够。

- [ ] **Step 1: shaders/gi/ibl/prefilter_diffuse.slang**

```hlsl
// shaders/gi/ibl/prefilter_diffuse.slang
[[vk::binding(0, 0)]] TextureCube<float4> gEnv;
[[vk::binding(1, 0)]] SamplerState         gLinear;
[[vk::binding(2, 0)]] RWTexture2DArray<float4> gDiffuseFaces;

[[vk::push_constant]] struct {
    uint outSize;
    uint _p0, _p1, _p2;
} gPC;

static const float PI = 3.14159265359;

float3 dirForFaceUV(uint face, float2 uv) {
    float2 st = uv * 2.0 - 1.0;
    float3 d;
    switch (face) {
        case 0: d = float3( 1.0, -st.y, -st.x); break;
        case 1: d = float3(-1.0, -st.y,  st.x); break;
        case 2: d = float3( st.x,  1.0,  st.y); break;
        case 3: d = float3( st.x, -1.0, -st.y); break;
        case 4: d = float3( st.x, -st.y,  1.0); break;
        default:d = float3(-st.x, -st.y, -1.0); break;
    }
    return normalize(d);
}

[shader("compute")]
[numthreads(8, 8, 1)]
void cs_main(uint3 dt : SV_DispatchThreadID) {
    if (dt.x >= gPC.outSize || dt.y >= gPC.outSize || dt.z >= 6) return;
    float2 uv = (float2(dt.xy) + 0.5) / float(gPC.outSize);
    float3 N = dirForFaceUV(dt.z, uv);

    float3 up    = abs(N.y) < 0.999 ? float3(0,1,0) : float3(1,0,0);
    float3 right = normalize(cross(up, N));
    up           = normalize(cross(N, right));

    float3 sum = float3(0);
    float weight = 0.0;
    const float dPhi = 0.025;
    const float dTheta = 0.025;
    for (float phi = 0.0; phi < 2.0 * PI; phi += dPhi) {
        for (float theta = 0.0; theta < 0.5 * PI; theta += dTheta) {
            float3 L_t = float3(sin(theta)*cos(phi), sin(theta)*sin(phi), cos(theta));
            float3 L = L_t.x * right + L_t.y * up + L_t.z * N;
            float3 c = gEnv.SampleLevel(gLinear, L, 0).rgb;
            sum += c * cos(theta) * sin(theta);
            weight += 1.0;
        }
    }
    float3 avg = PI * sum / weight;
    gDiffuseFaces[uint3(dt.xy, dt.z)] = float4(avg, 1.0);
}
```

## Task 8: prefilter_specular.slang + 调度（per-mip）

GGX 重要度采样：每个 mip 对应一个 roughness ∈ [0, 1]，用 Hammersley 序列产生 1024 samples。

- [ ] **Step 1: shaders/gi/ibl/prefilter_specular.slang**

```hlsl
// shaders/gi/ibl/prefilter_specular.slang
[[vk::binding(0, 0)]] TextureCube<float4> gEnv;
[[vk::binding(1, 0)]] SamplerState         gLinear;
[[vk::binding(2, 0)]] RWTexture2DArray<float4> gOutFaces;  // 当前 mip 的 array view

[[vk::push_constant]] struct {
    uint  outSize;       // 当前 mip 的边长
    uint  envCubeSize;   // env 第 0 mip 边长
    float roughness;
    uint  _p2;
} gPC;

static const float PI = 3.14159265359;

float radicalInverse_VdC(uint bits) {
    bits = (bits << 16u) | (bits >> 16u);
    bits = ((bits & 0x55555555u) << 1u) | ((bits & 0xAAAAAAAAu) >> 1u);
    bits = ((bits & 0x33333333u) << 2u) | ((bits & 0xCCCCCCCCu) >> 2u);
    bits = ((bits & 0x0F0F0F0Fu) << 4u) | ((bits & 0xF0F0F0F0u) >> 4u);
    bits = ((bits & 0x00FF00FFu) << 8u) | ((bits & 0xFF00FF00u) >> 8u);
    return float(bits) * 2.3283064365386963e-10;
}
float2 hammersley(uint i, uint N) {
    return float2(float(i) / float(N), radicalInverse_VdC(i));
}

float3 importanceSampleGGX(float2 Xi, float3 N, float a) {
    float phi = 2.0 * PI * Xi.x;
    float cosT = sqrt((1.0 - Xi.y) / (1.0 + (a*a - 1.0) * Xi.y));
    float sinT = sqrt(1.0 - cosT*cosT);
    float3 H_t = float3(cos(phi)*sinT, sin(phi)*sinT, cosT);

    float3 up    = abs(N.y) < 0.999 ? float3(0,1,0) : float3(1,0,0);
    float3 tan_  = normalize(cross(up, N));
    float3 bitan = cross(N, tan_);
    return normalize(tan_ * H_t.x + bitan * H_t.y + N * H_t.z);
}

float distributionGGX(float NoH, float a) {
    float a2 = a*a;
    float d = NoH*NoH * (a2 - 1.0) + 1.0;
    return a2 / (PI * d * d + 1e-7);
}

float3 dirForFaceUV(uint face, float2 uv) {
    float2 st = uv * 2.0 - 1.0;
    float3 d;
    switch (face) {
        case 0: d = float3( 1.0, -st.y, -st.x); break;
        case 1: d = float3(-1.0, -st.y,  st.x); break;
        case 2: d = float3( st.x,  1.0,  st.y); break;
        case 3: d = float3( st.x, -1.0, -st.y); break;
        case 4: d = float3( st.x, -st.y,  1.0); break;
        default:d = float3(-st.x, -st.y, -1.0); break;
    }
    return normalize(d);
}

[shader("compute")]
[numthreads(8, 8, 1)]
void cs_main(uint3 dt : SV_DispatchThreadID) {
    if (dt.x >= gPC.outSize || dt.y >= gPC.outSize || dt.z >= 6) return;
    float2 uv = (float2(dt.xy) + 0.5) / float(gPC.outSize);
    float3 N = dirForFaceUV(dt.z, uv);
    float3 V = N;

    const uint SAMPLE_COUNT = 1024u;
    float a = max(gPC.roughness * gPC.roughness, 1e-3);
    float3 prefiltered = float3(0);
    float totalWeight = 0.0;

    for (uint i = 0u; i < SAMPLE_COUNT; ++i) {
        float2 Xi = hammersley(i, SAMPLE_COUNT);
        float3 H = importanceSampleGGX(Xi, N, a);
        float3 L = normalize(2.0 * dot(V, H) * H - V);
        float NoL = saturate(dot(N, L));
        if (NoL > 0.0) {
            // 用 mip-bias 抑制亮度噪点（参考 LearnOpenGL IBL 章节）
            float NoH = saturate(dot(N, H));
            float D = distributionGGX(NoH, a);
            float pdf = (D * NoH / (4.0 * NoH)) + 1e-4;
            float saTexel = 4.0 * PI / (6.0 * gPC.envCubeSize * gPC.envCubeSize);
            float saSample = 1.0 / (float(SAMPLE_COUNT) * pdf + 1e-4);
            float mipLevel = (gPC.roughness == 0.0) ? 0.0 : 0.5 * log2(saSample / saTexel);

            prefiltered += gEnv.SampleLevel(gLinear, L, mipLevel).rgb * NoL;
            totalWeight += NoL;
        }
    }
    prefiltered = prefiltered / max(totalWeight, 1e-4);
    gOutFaces[uint3(dt.xy, dt.z)] = float4(prefiltered, 1.0);
}
```

```hlsl
// shaders/gi/ibl/brdf_lut.slang
[[vk::binding(0, 0)]] RWTexture2D<float2> gLut;

[[vk::push_constant]] struct {
    uint size;
    uint _p0, _p1, _p2;
} gPC;

static const float PI = 3.14159265359;

float radicalInverse_VdC(uint bits) {
    bits = (bits << 16u) | (bits >> 16u);
    bits = ((bits & 0x55555555u) << 1u) | ((bits & 0xAAAAAAAAu) >> 1u);
    bits = ((bits & 0x33333333u) << 2u) | ((bits & 0xCCCCCCCCu) >> 2u);
    bits = ((bits & 0x0F0F0F0Fu) << 4u) | ((bits & 0xF0F0F0F0u) >> 4u);
    bits = ((bits & 0x00FF00FFu) << 8u) | ((bits & 0xFF00FF00u) >> 8u);
    return float(bits) * 2.3283064365386963e-10;
}
float2 hammersley(uint i, uint N) {
    return float2(float(i) / float(N), radicalInverse_VdC(i));
}

float3 importanceSampleGGX(float2 Xi, float3 N, float a) {
    float phi = 2.0 * PI * Xi.x;
    float cosT = sqrt((1.0 - Xi.y) / (1.0 + (a*a - 1.0) * Xi.y));
    float sinT = sqrt(1.0 - cosT*cosT);
    float3 H_t = float3(cos(phi)*sinT, sin(phi)*sinT, cosT);
    float3 up    = abs(N.y) < 0.999 ? float3(0,1,0) : float3(1,0,0);
    float3 tan_  = normalize(cross(up, N));
    float3 bitan = cross(N, tan_);
    return normalize(tan_ * H_t.x + bitan * H_t.y + N * H_t.z);
}

float gSchlickIBL(float NoV, float a) {
    float k = (a*a) / 2.0;
    return NoV / (NoV * (1 - k) + k);
}
float gSmithIBL(float NoV, float NoL, float a) {
    return gSchlickIBL(NoV, a) * gSchlickIBL(NoL, a);
}

float2 integrateBRDF(float NoV, float roughness) {
    float3 V = float3(sqrt(1.0 - NoV*NoV), 0, NoV);
    float A = 0.0, B = 0.0;
    float3 N = float3(0, 0, 1);
    float a = max(roughness*roughness, 1e-3);
    const uint SAMPLES = 1024u;
    for (uint i = 0u; i < SAMPLES; ++i) {
        float2 Xi = hammersley(i, SAMPLES);
        float3 H = importanceSampleGGX(Xi, N, a);
        float3 L = normalize(2.0 * dot(V, H) * H - V);
        float NoL = saturate(L.z);
        float NoH = saturate(H.z);
        float VoH = saturate(dot(V, H));
        if (NoL > 0.0) {
            float G = gSmithIBL(NoV, NoL, a);
            float G_Vis = (G * VoH) / (NoH * NoV);
            float Fc = pow(1.0 - VoH, 5.0);
            A += (1.0 - Fc) * G_Vis;
            B += Fc * G_Vis;
        }
    }
    return float2(A, B) / float(SAMPLES);
}

[shader("compute")]
[numthreads(8, 8, 1)]
void cs_main(uint3 dt : SV_DispatchThreadID) {
    if (dt.x >= gPC.size || dt.y >= gPC.size) return;
    float NoV = (float(dt.x) + 0.5) / float(gPC.size);
    float roughness = (float(dt.y) + 0.5) / float(gPC.size);
    float2 ab = integrateBRDF(NoV, roughness);
    gLut[dt.xy] = ab;
}
```

```hlsl
[[vk::binding(0, 1)]] TextureCube<float4> gIblDiffuse;
[[vk::binding(1, 1)]] TextureCube<float4> gIblSpecular;
[[vk::binding(2, 1)]] Texture2D<float2>   gIblBrdfLut;
[[vk::binding(3, 1)]] SamplerState         gIblSampler;

float3 fresnelSchlickRoughness(float cosT, float3 F0, float roughness) {
    float a = 1.0 - roughness;
    return F0 + (max(float3(a), F0) - F0) * pow(saturate(1.0 - cosT), 5.0);
}

float3 evalIBL(float3 N, float3 V, float3 albedo, float metallic, float roughness,
               float specularMipCount)
{
    float NoV = saturate(dot(N, V));
    float3 R = reflect(-V, N);

    float3 F0 = lerp(float3(0.04), albedo, metallic);
    float3 F  = fresnelSchlickRoughness(NoV, F0, roughness);

    float3 kd = (1.0 - F) * (1.0 - metallic);
    float3 irradiance = gIblDiffuse.SampleLevel(gIblSampler, N, 0).rgb;
    float3 diffuse = irradiance * albedo;

    float mip = roughness * (specularMipCount - 1.0);
    float3 prefiltered = gIblSpecular.SampleLevel(gIblSampler, R, mip).rgb;
    float2 ab = gIblBrdfLut.SampleLevel(gIblSampler, float2(NoV, roughness), 0).xy;
    float3 specular = prefiltered * (F * ab.x + ab.y);

    return kd * diffuse + specular;
}
```

```hlsl
    VsOut o;
    float4 wp = mul(gPC.model, float4(i.pos, 1.0));
    o.worldPos = wp.xyz;
    o.svPos = mul(gFrame.viewProj, wp);
    float3x3 n3 = (float3x3)gPC.model;
    o.normal  = normalize(mul(n3, i.normal));
    o.tangent = float4(normalize(mul(n3, i.tangent.xyz)), i.tangent.w);
    o.uv = i.uv0;
    o.matIndex = gPC.materialIndex;
    return o;
}

float4 sampleTex(int idx, float2 uv) {
    if (idx < 0) return float4(1, 1, 1, 1);
    return gTextures[idx].Sample(gLinear, uv);
}

[shader("pixel")]
float4 ps_main(VsOut i) : SV_Target {
    MaterialGpu m = gMaterials[i.matIndex];

    float4 baseTex = sampleTex(m.baseColorTex, i.uv);
    float4 base = m.baseColorFactor * baseTex;
    if (m.alphaMode == 1u && base.a < m.alphaCutoff) discard;

    float metallic = m.metallicFactor;
    float roughness = m.roughnessFactor;
    if (m.mrTex >= 0) {
        float3 t = gTextures[m.mrTex].Sample(gLinear, i.uv).rgb;
        roughness *= t.g;
        metallic  *= t.b;
    }
    metallic = saturate(metallic);
    roughness = saturate(roughness);

    float3 N = normalize(i.normal);
    if (m.normalTex >= 0) {
        float3 nm = gTextures[m.normalTex].Sample(gLinear, i.uv).rgb * 2.0 - 1.0;
        nm.xy *= m.normalScale;
        float3 T = normalize(i.tangent.xyz);
        float3 B = normalize(cross(N, T) * i.tangent.w);
        N = normalize(T*nm.x + B*nm.y + N*nm.z);
    }

    float3 V = normalize(gFrame.cameraPos.xyz - i.worldPos);
    float3 L = normalize(-gFrame.sunDir.xyz);
    float3 sun = directBRDF(N, V, L, base.rgb, metallic, roughness) *
                 gFrame.sunColor_intensity.rgb * gFrame.sunColor_intensity.w;

    float occ = 1.0;
    if (m.occlusionTex >= 0) {
        float t = gTextures[m.occlusionTex].Sample(gLinear, i.uv).r;
        occ = lerp(1.0, t, m.occlusionStrength);
    }

    // IBL：specularMipCount 经 gFrame.counts.y 注入（约定 counts: x=matCount, y=specMips）
    float3 ibl = evalIBL(N, V, base.rgb, metallic, roughness,
                         (float)gFrame.counts.y);
    ibl *= occ;

    float3 emi = m.emissiveFactor;
    if (m.emissiveTex >= 0) emi *= gTextures[m.emissiveTex].Sample(gLinear, i.uv).rgb;

    return float4(sun + ibl + emi, base.a);
}
```

注意：`gFrame.iblSpecularMips` 是新字段，下一步加。

```cpp
void ForwardPass::init(Device& d, VkFormat colorFmt, VkFormat depthFmt, uint32_t maxTextures) {
    m_device = &d; m_colorFmt = colorFmt; m_depthFmt = depthFmt; m_maxTextures = maxTextures;

    // === Set=0 layout (与之前一致：UBO, SSBO, sampler, sampled image[]) ===
    std::array<VkDescriptorSetLayoutBinding, 4> b{};
    b[0] = {0, VK_DESCRIPTOR_TYPE_UNIFORM_BUFFER, 1,
            VK_SHADER_STAGE_VERTEX_BIT|VK_SHADER_STAGE_FRAGMENT_BIT};
    b[1] = {1, VK_DESCRIPTOR_TYPE_STORAGE_BUFFER, 1, VK_SHADER_STAGE_FRAGMENT_BIT};
    b[2] = {2, VK_DESCRIPTOR_TYPE_SAMPLER,        1, VK_SHADER_STAGE_FRAGMENT_BIT};
    b[3] = {3, VK_DESCRIPTOR_TYPE_SAMPLED_IMAGE,  maxTextures, VK_SHADER_STAGE_FRAGMENT_BIT};

    std::array<VkDescriptorBindingFlags, 4> bf{0, 0, 0,
        VK_DESCRIPTOR_BINDING_PARTIALLY_BOUND_BIT};
    VkDescriptorSetLayoutBindingFlagsCreateInfo bfci{
        VK_STRUCTURE_TYPE_DESCRIPTOR_SET_LAYOUT_BINDING_FLAGS_CREATE_INFO};
    bfci.bindingCount = (uint32_t)bf.size(); bfci.pBindingFlags = bf.data();

    VkDescriptorSetLayoutCreateInfo li{VK_STRUCTURE_TYPE_DESCRIPTOR_SET_LAYOUT_CREATE_INFO};
    li.pNext = &bfci;
    li.bindingCount = (uint32_t)b.size(); li.pBindings = b.data();
    VK_CHECK(vkCreateDescriptorSetLayout(d.device(), &li, nullptr, &m_setLayout));

    // === Descriptor pool + Set=0 alloc ===
    std::array<VkDescriptorPoolSize, 4> ps{{
        {VK_DESCRIPTOR_TYPE_UNIFORM_BUFFER, 1},
        {VK_DESCRIPTOR_TYPE_STORAGE_BUFFER, 1},
        {VK_DESCRIPTOR_TYPE_SAMPLER, 1},
        {VK_DESCRIPTOR_TYPE_SAMPLED_IMAGE, maxTextures},
    }};
    VkDescriptorPoolCreateInfo pci{VK_STRUCTURE_TYPE_DESCRIPTOR_POOL_CREATE_INFO};
    pci.maxSets = 1; pci.poolSizeCount = (uint32_t)ps.size(); pci.pPoolSizes = ps.data();
    VK_CHECK(vkCreateDescriptorPool(d.device(), &pci, nullptr, &m_pool));

    VkDescriptorSetAllocateInfo dai{VK_STRUCTURE_TYPE_DESCRIPTOR_SET_ALLOCATE_INFO};
    dai.descriptorPool = m_pool; dai.descriptorSetCount = 1; dai.pSetLayouts = &m_setLayout;
    VK_CHECK(vkAllocateDescriptorSets(d.device(), &dai, &m_set));

    m_frameUbo = Buffer(d, sizeof(FrameUBO),
                       VK_BUFFER_USAGE_UNIFORM_BUFFER_BIT,
                       VK_MEMORY_PROPERTY_HOST_VISIBLE_BIT | VK_MEMORY_PROPERTY_HOST_COHERENT_BIT);
}

void ForwardPass::buildPipeline(const char* variant, VkDescriptorSetLayout giDsl) {
    auto& d = *m_device;

    VkPushConstantRange pc{};
    pc.stageFlags = VK_SHADER_STAGE_VERTEX_BIT | VK_SHADER_STAGE_FRAGMENT_BIT;
    pc.size = 64 + 16;

    std::array<VkDescriptorSetLayout, 2> sets{m_setLayout, giDsl};

    VkPipelineLayoutCreateInfo plci{VK_STRUCTURE_TYPE_PIPELINE_LAYOUT_CREATE_INFO};
    plci.setLayoutCount = giDsl ? 2 : 1; plci.pSetLayouts = sets.data();
    plci.pushConstantRangeCount = 1; plci.pPushConstantRanges = &pc;
    VK_CHECK(vkCreatePipelineLayout(d.device(), &plci, nullptr, &m_pipelineLayout));

    auto sd = shaderDir();
    std::string spvName = std::string("forward_") + variant + ".spv";
    if (std::strcmp(variant, "default") == 0) spvName = "forward.spv";
    ShaderModule vs(d, sd / "forward" / spvName);
    ShaderModule fs(d, sd / "forward" / spvName);

    VkPipelineShaderStageCreateInfo stages[2]{};
    stages[0] = {VK_STRUCTURE_TYPE_PIPELINE_SHADER_STAGE_CREATE_INFO};
    stages[0].stage = VK_SHADER_STAGE_VERTEX_BIT;
    stages[0].module = vs.handle(); stages[0].pName = "vs_main";
    stages[1] = {VK_STRUCTURE_TYPE_PIPELINE_SHADER_STAGE_CREATE_INFO};
    stages[1].stage = VK_SHADER_STAGE_FRAGMENT_BIT;
    stages[1].module = fs.handle(); stages[1].pName = "ps_main";

    VkVertexInputBindingDescription vib{0, sizeof(Vertex), VK_VERTEX_INPUT_RATE_VERTEX};
    std::array<VkVertexInputAttributeDescription, 4> via{};
    via[0] = {0, 0, VK_FORMAT_R32G32B32_SFLOAT,    offsetof(Vertex,position)};
    via[1] = {1, 0, VK_FORMAT_R32G32B32_SFLOAT,    offsetof(Vertex,normal)};
    via[2] = {2, 0, VK_FORMAT_R32G32B32A32_SFLOAT, offsetof(Vertex,tangent)};
    via[3] = {3, 0, VK_FORMAT_R32G32_SFLOAT,       offsetof(Vertex,uv0)};

    VkPipelineVertexInputStateCreateInfo vi{VK_STRUCTURE_TYPE_PIPELINE_VERTEX_INPUT_STATE_CREATE_INFO};
    vi.vertexBindingDescriptionCount = 1; vi.pVertexBindingDescriptions = &vib;
    vi.vertexAttributeDescriptionCount = (uint32_t)via.size();
    vi.pVertexAttributeDescriptions = via.data();

    VkPipelineInputAssemblyStateCreateInfo ia{VK_STRUCTURE_TYPE_PIPELINE_INPUT_ASSEMBLY_STATE_CREATE_INFO};
    ia.topology = VK_PRIMITIVE_TOPOLOGY_TRIANGLE_LIST;

    VkPipelineViewportStateCreateInfo vp{VK_STRUCTURE_TYPE_PIPELINE_VIEWPORT_STATE_CREATE_INFO};
    vp.viewportCount = 1; vp.scissorCount = 1;

    VkPipelineRasterizationStateCreateInfo rs{VK_STRUCTURE_TYPE_PIPELINE_RASTERIZATION_STATE_CREATE_INFO};
    rs.cullMode = VK_CULL_MODE_BACK_BIT;
    rs.frontFace = VK_FRONT_FACE_COUNTER_CLOCKWISE;
    rs.lineWidth = 1.0f;

    VkPipelineMultisampleStateCreateInfo ms{VK_STRUCTURE_TYPE_PIPELINE_MULTISAMPLE_STATE_CREATE_INFO};
    ms.rasterizationSamples = VK_SAMPLE_COUNT_1_BIT;

    VkPipelineDepthStencilStateCreateInfo ds{VK_STRUCTURE_TYPE_PIPELINE_DEPTH_STENCIL_STATE_CREATE_INFO};
    ds.depthTestEnable = VK_TRUE; ds.depthWriteEnable = VK_TRUE;
    ds.depthCompareOp = VK_COMPARE_OP_LESS_OR_EQUAL;

    VkPipelineColorBlendAttachmentState ba{}; ba.colorWriteMask = 0xF;
    VkPipelineColorBlendStateCreateInfo cb{VK_STRUCTURE_TYPE_PIPELINE_COLOR_BLEND_STATE_CREATE_INFO};
    cb.attachmentCount = 1; cb.pAttachments = &ba;

    VkDynamicState dyn[] = {VK_DYNAMIC_STATE_VIEWPORT, VK_DYNAMIC_STATE_SCISSOR};
    VkPipelineDynamicStateCreateInfo dyni{VK_STRUCTURE_TYPE_PIPELINE_DYNAMIC_STATE_CREATE_INFO};
    dyni.dynamicStateCount = 2; dyni.pDynamicStates = dyn;

    VkPipelineRenderingCreateInfo rci{VK_STRUCTURE_TYPE_PIPELINE_RENDERING_CREATE_INFO};
    rci.colorAttachmentCount = 1; rci.pColorAttachmentFormats = &m_colorFmt;
    rci.depthAttachmentFormat = m_depthFmt;

    VkGraphicsPipelineCreateInfo gpci{VK_STRUCTURE_TYPE_GRAPHICS_PIPELINE_CREATE_INFO};
    gpci.pNext = &rci;
    gpci.stageCount = 2; gpci.pStages = stages;
    gpci.pVertexInputState = &vi; gpci.pInputAssemblyState = &ia;
    gpci.pViewportState = &vp; gpci.pRasterizationState = &rs;
    gpci.pMultisampleState = &ms; gpci.pDepthStencilState = &ds;
    gpci.pColorBlendState = &cb; gpci.pDynamicState = &dyni;
    gpci.layout = m_pipelineLayout;
    VK_CHECK(vkCreateGraphicsPipelines(d.device(), VK_NULL_HANDLE, 1, &gpci, nullptr, &m_pipeline));
}

void ForwardPass::destroyPipeline() {
    if (!m_device) return;
    if (m_pipeline) vkDestroyPipeline(m_device->device(), m_pipeline, nullptr);
    if (m_pipelineLayout) vkDestroyPipelineLayout(m_device->device(), m_pipelineLayout, nullptr);
    m_pipeline = VK_NULL_HANDLE; m_pipelineLayout = VK_NULL_HANDLE;
}

void ForwardPass::setTechnique(IGITechnique* tech) {
    m_tech = tech;
    destroyPipeline();
    buildPipeline(tech ? tech->shaderVariant() : "default",
                  tech ? tech->descriptorSetLayout() : VK_NULL_HANDLE);
}

void ForwardPass::destroy() {
    destroyPipeline();
    auto dev = m_device->device();
    if (m_pool) vkDestroyDescriptorPool(dev, m_pool, nullptr);
    if (m_setLayout) vkDestroyDescriptorSetLayout(dev, m_setLayout, nullptr);
    m_frameUbo.reset();
}
```

## 风险 / 退路

| 风险 | 退路 |
|---|---|
| Slang 不支持某语法（例如 SamplerState/TextureCube 在 set=1） | 回退到 GLSL；ibl_eval 改写为 `samplerCube` GLSL，slang 编译时按 stage 拆分 |
| 烘焙时间过长（>5s） | 把 SAMPLE_COUNT 从 1024 降到 256；envCubeSize 从 512 降到 256 |
| Sponza IBL 太暗 | 检查 stbi_loadf 是否正确得到 linear（HDR 本身就 linear）；检查 cubemap 像素值是否 > 1.0（亮区） |
| stbi_loadf 不识别某些 HDR | 替换为 `.exr` + tinyexr，或确保用 Radiance .hdr |
| 切换 GI 时 pipeline 重建造成 1 帧黑屏 | 在 setTechnique 前 vkDeviceWaitIdle；M3 只在启动 attach 一次，不影响 |
